document.addEventListener("DOMContentLoaded", function () {
  // ページが開いたら行いたい処理
  Load().then(() => {
    return printDater();
  });
});

const stockList=[];



function Load() {
  const dater = loadCSV("schedule.csv");

  // ↓ ここに return を追加する！
  return dater.then(data => {
    console.log(data[0][0]); // ID
    console.log(data[0][1]); // 名前

    for (let i = 0; i < data.length; i++) {
      const adata = [];
      for (let a = 0; a < 10; a++) {
        adata.push(data[i][a]);
      }

      stockList.push(adata);
    }
  });
}
```

流れ
1. 起動
2. Load()を実行
   1. csvを読み込む
   2. 配列に格納
3. printDater()を実行


注意事項1 

    ```
    時間がかかるメソッド().then(()=>{
        return 待ってるメソッド();
    })

    Load().then(() => {
        return printDater();
    });
